### 3.2 منطق حساب الأوقات المتاحة

```javascript
async function getAvailableSlots(branchId, date, serviceId, staffId) {
  // 1. جلب أوقات العمل للفرع في هذا اليوم
  const workingHours = await getWorkingHours(branchId, date);
  if (!workingHours.isOpen) {
    return { available: false, reason: 'الصالون مغلق' };
  }
  
  // 2. جلب العطلات
  const isHoliday = await checkHoliday(branchId, date);
  if (isHoliday) {
    return { available: false, reason: 'عطلة' };
  }
  
  // 3. جلب الحجوزات الموجودة
  const existingBookings = await getBookings(branchId, date, staffId);
  
  // 4. حساب الأوقات المتاحة
  const slots = [];
  let currentTime = workingHours.open;
  
  while (currentTime < workingHours.close) {
    const endTime = addMinutes(currentTime, service.duration_min);
    
    // التحقق من عدم التعارض مع الحجوزات الموجودة
    const isAvailable = !existingBookings.some(booking => 
      timesOverlap(currentTime, endTime, booking.start_time, booking.end_time)
    );
    
    if (isAvailable) {
      slots.push({
        start: currentTime,
        end: endTime,
        available: true
      });
    }
    
    currentTime = addMinutes(currentTime, 30); // فاصل 30 دقيقة
  }
  
  return { available: true, slots };
}
```

### 3.4 Edge Cases

- **إلغاء الحجز**: 
  - قبل 24 ساعة: استرجاع كامل
  - بعد 24 ساعة: خصم 50%
  - عدم الحضور (no-show): خصم كامل
  
- **تعديل الحجز**: 
  - تغيير الوقت: تحقق من التوفر
  - تغيير الموظف: تحقق من التخصص
  - تغيير الخدمة: إعادة حساب السعر

- **تذكيرات**:
  - قبل 24 ساعة: إشعار
  - قبل ساعة: إشعار
  - قبل 15 دقيقة: إشعار

### 4.3 منطق التحقق من المخزون

```javascript
async function validateCart() {
  const cart = getCart();
  
  for (const item of cart.items) {
    const product = await getProduct(item.product_id);
    
    // التحقق من التوفر
    if (!product.is_available) {
      return {
        valid: false,
        error: `المنتج "${product.name}" غير متوفر`,
        action: 'remove'
      };
    }
    
    // التحقق من الكمية
    if (product.stock_quantity < item.quantity) {
      return {
        valid: false,
        error: `الكمية المطلوبة من "${product.name}" غير متوفرة`,
        available: product.stock_quantity,
        action: 'update_quantity'
      };
    }
    
    // التحقق من السعر (قد يكون تغير)
    if (product.price !== item.price) {
      return {
        valid: false,
        error: `سعر "${product.name}" تغير`,
        new_price: product.price,
        action: 'update_price'
      };
    }
  }
  
  return { valid: true };
}
```

| عنوان خارج النطاق | "لا يمكن التوصيل لهذا العنوان" | تعديل العنوان |
| حد أدنى للطلب | "الحد الأدنى للطلب 100 MAD" | إضافة منتجات |
| محفظة فارغة | "رصيد المحفظة غير كافٍ" | شحن المحفظة |
| كوبون منتهي | "الكوبون منتهي الصلاحية" | إزالة الكوبون |
